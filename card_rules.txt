    function escapeHtml(value) {
      return String(value)
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&#039;");
    }

    function renderTrash() {
      renderTrashSide("player", elements.playerTrash);
      renderTrashSide("opponent", elements.opponentTrash);
      if (state.battleInspector?.kind === "trash") renderBattleInspector();
    }

    function renderTrashSide(side, container) {
      const trash = state.players[side].trash;
      const box = container.closest(".trash-box");
      if (box) box.dataset.trashCount = String(trash.length);
      const counts = new Map();
      trash.forEach((card) => counts.set(card.name, (counts.get(card.name) || 0) + 1));
      const ownerTone = localTrashOwnerTone(side);
      container.innerHTML = `<div class="trash-total">合計 ${trash.length}枚</div>`;
      const list = document.createElement("div");
      list.className = "trash-card-list";
      if (trash.length === 0) {
        list.textContent = "カードなし";
      } else {
        [...counts.entries()]
          .sort((a, b) => a[0].localeCompare(b[0], "ja"))
          .forEach(([name, count]) => {
            const row = document.createElement("div");
            row.className = `trash-card-row trash-owner-${ownerTone}`;
            row.innerHTML = `<span>${escapeHtml(name)}</span><span class="trash-card-count">×${count}</span>`;
            list.appendChild(row);
          });
      }
      container.appendChild(list);
    }

      state.battleInspector = null;
      const inspector = document.getElementById("battleDrawerInspector");
      const title = document.getElementById("battleDrawerTitle");

    }
